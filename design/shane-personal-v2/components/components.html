HTML — note the .s-edge trap 
<!-- A hatched rect MUST use .s-edge, not .s-ink.
     .s-ink declares fill:none, and CSS beats the presentation
     attribute — which silently erases the hatch fill. -->
<rect fill="url(#hatch)" class="s-edge"/>

<!-- pattern contents can't inherit currentColor,
     so the hatch line is styled by class -->
<pattern id="hatch" width="7" height="7" patternUnits="userSpaceOnUse"
